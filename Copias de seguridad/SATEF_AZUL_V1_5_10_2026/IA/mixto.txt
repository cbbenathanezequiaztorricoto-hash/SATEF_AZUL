@app.route('/conductor/editar/<int:id_conductor>', methods=['POST'])
def editar_conductor(id_conductor):
    if 'usuario' not in session: return redirect(url_for('index'))
    conn = get_db_connection()
    if conn:
        cursor = conn.cursor()
        # .get() evita que el servidor caiga si un campo no viene en la petición
        primer_nombre = request.form.get('primerNombreConductor', '')
        segundo_nombre = request.form.get('segundoNombreConductor', '')
        apellido_paterno = request.form.get('apellidoPaterno', '')
        apellido_materno = request.form.get('apellidoMaterno', '')
        ci = request.form.get('ci', '')
        telf = request.form.get('telf', '')
        licencia = request.form.get('licencia', '')
        try:
            cursor.execute("""
            UPDATE Conductor 
            SET primerNombreConductor = %s,
                segundoNombreConductor = %s,
                apellidoPaterno = %s,
                apellidoMaterno = %s,
                ci = %s,
                telf = %s,
                categoriaLicenciaConducir = %s
            WHERE idConductor = %s
            """, (primer_nombre, segundo_nombre, apellido_paterno, apellido_materno, ci, telf, licencia, id_conductor))
            conn.commit()
        except Error as e:
            print(f"Error al editar conductor: {e}")
        finally:
            cursor.close()
            conn.close()
    return redirect(url_for('dashboard'))
